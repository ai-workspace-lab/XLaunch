cmake_minimum_required(VERSION 3.24)
project(XLaunch VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Widgets)
qt_standard_project_setup(REQUIRES 6.8)
if(APPLE)
  enable_language(OBJCXX)
  set(PLATFORM_SOURCE src/platform_macos.mm)
else()
  set(PLATFORM_SOURCE src/platform_stub.cpp)
endif()
qt_add_executable(XLaunch MACOSX_BUNDLE src/main.cpp src/catalog.cpp src/catalog.h src/platform.h ${PLATFORM_SOURCE})
qt_add_qml_module(XLaunch URI XLaunch QML_FILES qml/Main.qml RESOURCES assets/amber-wallpaper.png)
target_link_libraries(XLaunch PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets)
if(APPLE)
  target_link_libraries(XLaunch PRIVATE "-framework AppKit")
  set_target_properties(XLaunch PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.aiworkspacelab.xlaunch MACOSX_BUNDLE_BUNDLE_NAME XLaunch MACOSX_BUNDLE_SHORT_VERSION_STRING 0.1.0)
endif()
enable_testing()
add_test(NAME catalog-smoke COMMAND XLaunch --self-test)
